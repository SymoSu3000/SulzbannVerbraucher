function temperatureText( data ) { if ( data.temperature === null || data.temperature === undefined || data.temperatureLimit === null || data.temperatureLimit === undefined ) { return "Max. Temp."; } const actual = Number( data.temperature ); const limit = Number( data.temperatureLimit ); if ( Number.isNaN(actual) || Number.isNaN(limit) ) { return "Max. Temp."; } return ( "Ist " + actual.toFixed(1) + " °C · Max " + limit.toFixed(0) + " °C" ); }
